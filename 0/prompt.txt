```ts
const NODE_MIN_WIDTH = 200;
const NODE_MAX_WIDTH = 320;
const HEADER_HEIGHT = 56;   // two lines
const TITLE_CHAR_WIDTH = 7.4;   // 13px / 600 weight
const META_CHAR_WIDTH = 5.9;    // 11px / 400 weight
const TITLE_CHROME_WIDTH = 84;
const META_CHROME_WIDTH = 48;

function nodeWidth(label: string, metaLabel: string): number {
	return Math.round(Math.min(NODE_MAX_WIDTH, Math.max(
		NODE_MIN_WIDTH,
		TITLE_CHROME_WIDTH + label.length * TITLE_CHAR_WIDTH,
		META_CHROME_WIDTH + metaLabel.length * META_CHAR_WIDTH,
	)));
}

function schemaMetaLabel(fieldCount: number, entityCount: number | undefined): string {
	const parts: string[] = [];
	if (fieldCount > 0) parts.push(`${fieldCount} ${fieldCount === 1 ? "field" : "fields"}`);
	if (typeof entityCount === "number")
		parts.push(`${entityCount.toLocaleString()} ${entityCount === 1 ? "record" : "records"}`);
	return parts.join(" · ");
}
```

```tsx
export function SchemaTypeEditorPanel({ slug, onClose }: {
	slug: string | null;   // null = create mode
	onClose: () => void;
}) {
	const { orgContext } = useOrgContext();
	const { data: entityTypes = [], isLoading } = useEntityTypesAdmin(orgContext);
	if (slug === null)
		return <EntityTypePage entityType={null} orgContext={orgContext} onClose={onClose} compact />;
	if (isLoading) return <TabLoadingState label="entity type" variant="panel" />;

	if (!entityType) return <p className="text-sm text-muted-foreground">Entity type not found.</p>;

		hideDelete={isSystemEntityType(entityType)} onClose={onClose} compact />;
}
```
