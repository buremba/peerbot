/** Slack mrkdwn only — Markdown does NOT decode HTML entities. */
function escapeSlackText(value: string): string {
	return value.replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;");
}

/**
 * Escape user/agent-controlled text for the in-app Markdown body (GFM...).
 */
function escapeMarkdownText(value: string): string {
	return value
		.replace(/([\\`*_~[\]<>])/g, "\\$1")
		// Setext underlines of ANY run length ("Trusted line\n=" renders <h1>)
		.replace(/^([ \t]{0,3})([-=]+)(?=[ \t]*$)/gm, "$1\\$2")
		.replace(/^([ \t]{0,3})(?=[#>+-](?:\s|$))/gm, "$1\\")
		.replace(/^([ \t]{0,3}\d{1,9})([.)])(?=\s|$)/gm, "$1\\$2");
}

/** Raw display text; each emitter applies its own escaping. */
function displayNotificationValue(value: unknown): string {
	if (value === undefined || value === null || value === "") return "Not set";
	if (typeof value === "string") return value;
	return JSON.stringify(value, null, 2);
}

### `packages/owletto/src/lib/notification-summary.ts` (NEW)
```ts
const REVIEW_LINK_LINE = /^\[Review in Lobu\]\(.*\)$/;
const DETAIL_LINE = /^[-*]\s/;

export function notificationSummaryMarkdown(body: string | null | undefined): string | null {
	if (!body) return null;
	const summary: string[] = [];
	for (const raw of body.split("\n")) {
		const line = raw.trim();
		if (line === "") { if (summary.length > 0) break; continue; }
		if (DETAIL_LINE.test(line) || REVIEW_LINK_LINE.test(line)) break;
		summary.push(line);
	}
	const text = summary.join(" ").trim();
	if (text !== "") return text;
	// Fall back to leading lines (minus review CTA) so a bullet-only agent digest
	// still shows a body instead of rendering blank.
	const fallback = body.split("\n").map(l => l.trim())
		.filter(l => l !== "" && !REVIEW_LINK_LINE.test(l)).join(" ").trim();
	return fallback === "" ? null : fallback;
}
```

```ts
function isOwnAppHref(href: string | undefined): boolean {
	if (!href) return false;
	try {
		const base = typeof window === "undefined" ? "https://localhost" : window.location.href;
		const url = new URL(href, base);
		if (url.protocol !== "https:" && url.protocol !== "http:") return false;
		return typeof window === "undefined" || url.host === window.location.host;
	} catch { return false; }
}

const components: Components = {
	// ...
	a: ({ href, children }) =>
		isOwnAppHref(href) ? (<a href={href} target="_blank" rel="noopener noreferrer"
			className="font-medium text-primary underline underline-offset-2">{children}</a>)
			: (<>{children}</>),
	img: ({ alt }) => <>{alt}</>,
};

const inlineComponents: Components = { ...components,
	p/h1-h6/ul/ol/li/blockquote/pre: ({children}) => <>{children}</>, hr: () => null };
const inlineComponentsNoLinks: Components = { ...inlineComponents, a: ({children}) => <>{children}</> };

export const InlineMarkdownText: FC<{ text: string; className?: string;
	linkPolicy?: "app-links" | "strip" }> = ({ text, className, linkPolicy = "strip" }) => (
	<span className={cn("break-words", className)}>
		<ReactMarkdown remarkPlugins={[remarkGfm]}
			components={linkPolicy === "app-links" ? inlineComponents : inlineComponentsNoLinks}>
			{text}
		</ReactMarkdown>
	</span>
);
```
