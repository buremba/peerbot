```ts
	// Validate metadata against entity type's JSON schema (if being updated)
	if (args.metadata !== undefined && Object.keys(args.metadata).length > 0) {
		/*
		 * A top-level null is the update contract's explicit clear sentinel, not a
		 * value, so it must not reach the validator as one. Passing it through had
		 * two failure modes, both silent from the caller's side: on an optional
		 * typed property the clear was REJECTED ("status: must be one of ..."),
		 * and on a required property AJV's coercion turned null into "" and wrote
		 * the empty string, so a clear that should have been refused corrupted the
		 * row instead. Filtering nulls out makes a clear read as an absent
		 * property, which is exactly what it means: optional clears validate,
		 * required clears fail as a missing required property.
		 *
		 * Validating a filtered COPY rather than `args.metadata` itself is
		 * deliberate: the field merge below still needs the nulls to know which
		 * keys to clear. But AJV coerces its input in place, so every non-null
		 * coercion has to be copied back afterwards or the write silently stores
		 * the uncoerced value.
		 */
		const metadataForValidation = Object.fromEntries(
			Object.entries(args.metadata).filter(([, value]) => value !== null),
		);
		const validation = await validateEntityMetadata(
			before.entity_type as string,
			metadataForValidation,
			ctx,
		);
		if (!validation.valid) { ... throw new ToolUserError(...); }
		Object.assign(args.metadata, metadataForValidation);
	}
```
