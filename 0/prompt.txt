```ts
/**
 * A merged Owletto PR that changes nothing outside `deploy/` has no visual
 * surface, so there is no before/after to capture and a screenshot pair would
 * be theatre. `permittedFluxTailParent` cannot cover this: the tail walk in
 * `resolveOwlettoPullRequest` returns as soon as a commit matches a merged PR,
 * so a human-authored deploy-only PR at the pointer never reaches it.
 *
 * Deliberately NOT keyed on author or subject. Provenance is already
 * established — this is a merged PR in the Owletto repo, reviewed as code. The
 * only question here is whether it has a UI to prove, and the file paths answer
 * that on their own.
 *
 * An empty list is NOT deploy-only: the caller must pass a fully paginated file
 * list, and an empty one means the fetch told us nothing rather than that the
 * PR is safe.
 */
export function isDeployOnlyPullRequest(
  files: Array<{ filename: string }>
): boolean {
  if (files.length === 0) return false;
  return files.every((file) => file.filename.startsWith("deploy/"));
}
```
