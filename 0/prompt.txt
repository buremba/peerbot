       const sections: string[] = [];
       if (layers.identityMd.trim()) sections.push(`## Agent Identity\n\n${layers.identityMd.trim()}`);
       if (layers.soulMd.trim()) sections.push(`## Agent Instructions\n\n${layers.soulMd.trim()}`);
       if (layers.userMd.trim()) sections.push(`## User Context\n\n${layers.userMd.trim()}`);
       const composed = sections.join("\n\n");
       return composed || layers.unconfiguredNotice;
     }
     export function resolveAgentIdentity(layers: AgentInstructionLayers | undefined): string {
       const composed = layers ? composeAgentInstructions(layers).trim() : "";
       return composed.length > 0 ? composed : LOBU_DEFAULT_IDENTITY;
     }
