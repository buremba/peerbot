       response: Response
     ): Promise<void> {

       try {
         if (response.ok) { await clearInferenceProviderError(organizationId, slug); return; }
         const code =
           response.status === 429 || response.status === 402
             ? AgentErrorCode.PROVIDER_QUOTA_EXHAUSTED
             : response.status === 401 || response.status === 403
               ? AgentErrorCode.PROVIDER_AUTH
               : undefined;
         if (!code) return;
         const retryAfter = response.headers.get("retry-after");
         await markInferenceProviderUnhealthy(
           organizationId, slug,
           `HTTP ${response.status}${retryAfter ? ` (retry-after: ${retryAfter})` : ""} — ${code}`
         );
       } catch (err) { logger.warn(`Failed to record provider health for ${slug}: ...`); }
     }
