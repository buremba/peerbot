```tsx
/** `2026-08-01` — a date with no time and therefore no timezone. */
const DATE_ONLY = /^\d{4}-\d{2}-\d{2}$/;

/**
 * A date-only string carries no timezone, so `new Date("2026-08-01")` — which

function parseSchemaDate(value: string): Date {
	if (DATE_ONLY.test(value)) {
		const [year, month, day] = value.split("-").map(Number);

}
```

**`packages/owletto/src/components/entity-tabs/index.tsx`** — `buildMetadataPatch` rewritten:
```tsx
/**
 * Sends the WHOLE current object, not just the changed keys. The server
 * validates the patch as a document in its own right rather than the merge of
 * patch and stored row, so a patch carrying only the edited field fails schema
 * validation the moment the type declares anything `required`...
 */
export function buildMetadataPatch(
	current: Record<string, unknown>,
	baseline: Record<string, unknown>,
): Record<string, unknown> | undefined {
	const keys = new Set([...Object.keys(current), ...Object.keys(baseline)]);
	let dirty = false;
	const patch: Record<string, unknown> = {};
	for (const key of keys) {
		const hasCurrent = key in current;
		const hasBaseline = key in baseline;
		if (!hasCurrent && hasBaseline) { patch[key] = null; dirty = true; continue; }
		if (hasCurrent) {
			patch[key] = current[key];
			if (!hasBaseline || JSON.stringify(current[key]) !== JSON.stringify(baseline[key])) dirty = true;
		}
	}
	return dirty ? patch : undefined;
}
```
