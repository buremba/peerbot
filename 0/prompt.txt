   **`packages/owletto/src/lib/behavior-skills.ts`** (NEW, owletto#634) — the web mirror of the CLI compile. Central to the whole rework.
   ```ts
   export const MAX_BEHAVIOR_SKILLS = 5;
   export const MAX_COMPILED_INSTRUCTIONS_BYTES = 32 * 1024;
   export function skillBody(skill: BehaviorSkillOption): string {
     return (skill.content ?? skill.instructions ?? "").trim();
   }
   export function behaviorRequiresSkills(triggers: BehaviorTriggerLike[] | undefined): boolean {
     const list = triggers ?? [];
     if (list.length === 0) return true;
     return list.some((t) => t.kind === "schedule" || t.execution === "window");
   }
   export function compileBehaviorInstructions(selected: string[], library: BehaviorSkillOption[]): string {
     if (selected.length === 0) return "";
     const byName = new Map(library.map((s) => [s.name, s]));
     const bodies: string[] = [];
     for (const name of selected) {
       const skill = byName.get(name);
       const body = skill ? skillBody(skill) : "";
       if (body) bodies.push(body);
     }
     return bodies.join("\n\n");
   }
   export function compiledByteLength(compiled: string): number {
     return new TextEncoder().encode(compiled).length;
   }
   ```

   ```ts
   /**
    * Mark a server-dispatched Behavior run completed, recording who executed it.
    *
    * `complete_window` defaults `runs.model_used` to the literal 'external-client'
    * whenever its caller omits `model` (see complete-window.ts), and the platform's
    * own Lobu agent omits it — so a run the SERVER dispatched ends up labelled as
    * though an outside MCP client had executed it. That label reads as an
    * observation and is not one: triaging the July 2026 Behavior collapse from this
    * column produced exactly that wrong conclusion.
    *
    * This path is reached only for runs the server dispatched to the Lobu agent —
    * it is keyed on `dispatched_message_id`, which only `watchers/automation.ts`
    * sets — so the executor is known here. Mirrors the device-worker stamp in
    * `worker-api/run-lifecycle.ts`. A model the agent DID report always wins; this
    * only replaces the placeholder.
    *
    * (SQL comments are kept out of the template literal: a backtick inside one
    * terminates the string and fails the esbuild transform, not just at runtime.)
    */
   export async function markWatcherRunCompleted(
     sql: DbClient, runId: number, windowId: number | null
   ): Promise<void> {
     await sql`
       UPDATE runs
       SET status = 'completed',
           window_id = ${windowId},
           completed_at = current_timestamp,
           error_message = NULL,
           model_used = CASE
             WHEN model_used = 'external-client' OR model_used IS NULL
               THEN 'lobu-agent'
             ELSE model_used
           END
       WHERE id = ${runId}
         AND status = ANY(${runStatusLiteral(ACTIVE_RUN_STATUSES)}::text[])
     `;
   }
   ```
