- **Per-org subdomain vs path-based routing**: `{org}.{zone}` Host-header extraction vs `/{org}` path segments
- **Flux CD ImagePolicy**: `filterTags.pattern` + `extract`, policy types `numerical` (uses `strconv.ParseFloat(version, 64)` — float64, ~15-17 significant digits) vs `alphabetical` (string sort, no precision ceiling)
- **GitHub Actions concurrency groups**: only ONE pending run per group; a third arrival evicts the queued second, silently (cancelled runs report no failure)

```ts
export function injectRuntimeConfig(templateHtml: string): string {
  const zone = getConfiguredSubdomainZone();
  const script = `<script>window.__LOBU_RUNTIME_CONFIG__=${serializeForScript({
    subdomainZone: zone,
  })};</script>`;
```

```ts
export function getConfiguredSubdomainZone(
  cookieDomain = process.env.AUTH_COOKIE_DOMAIN
): string | null {
  const zone = normalizeHost(cookieDomain);
  return zone && isSubdomainCapableHost(zone) ? zone : null;
}
function isSubdomainCapableHost(host: string): boolean {
  if (LOCALHOST_HOSTNAMES.has(host)) return false;
  if (/^\d{1,3}(\.\d{1,3}){3}$/.test(host)) return false;
  if (host.includes(':') || host.startsWith('[')) return false;
  return true;
}
```
