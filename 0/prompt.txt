     ```typescript
     /** Recent threads across all agents in a workspace (sidebar). */
     export function useRecentAgentThreads(
     	orgSlug?: string | null,
     	agentIds?: string[],
     	options?: { enabled?: boolean; limit?: number },
     ) {
     	const limit = options?.limit ?? 8;
     	const enabled =
     		!!orgSlug && (agentIds?.length ?? 0) > 0 && (options?.enabled ?? true);

     	const queries = useQueries({
     		queries: (agentIds ?? []).map((agentId) => ({
     			...agentThreadsQueryOptions(orgSlug, agentId, {
     				strict: false,
     				scope: "all" as const,
     			}),
     			enabled,
     		})),
     	});

     	const isLoading = enabled && queries.some((q) => q.isLoading);
     	const threads = queries
     		.flatMap((q, index) => {
     			const agentId = agentIds?.[index];
     			if (!agentId) return [];
     			return (q.data ?? []).map((thread) => ({ agentId, ...thread }));
     		})
     		.sort((a, b) => b.updatedAt - a.updatedAt)
     		.slice(0, limit);

     	return { threads, isLoading };
     }
     ```

     	orgSlug,
